def obter_estabelecimentos():
    """Retorna lista de todos os estabelecimentos"""
    return ESTABELECIMENTOS_POA


def obter_por_tipo(tipo: str):
    """
    Retorna estabelecimentos filtrados por tipo

    Args:
        tipo: "hospital", "upa", "ubs"

    Returns:
        List[dict]: Estabelecimentos do tipo especificado
    """
    return [e for e in ESTABELECIMENTOS_POA if e["tipo"] == tipo]


def obter_por_id(est_id: int):
    """
    Retorna um estabelecimento específico por ID

    Args:
        est_id: ID do estabelecimento

    Returns:
        dict or None: Estabelecimento encontrado ou None
    """
    for est in ESTABELECIMENTOS_POA:
        if est["id"] == est_id:
            return est
    return None
